                步驟 2
                貼上後端同步腳本
              

              
清空 Apps Script 編輯器內原本文字，點擊下方「一鍵複製」按鈕，貼入編輯器後按儲存（Ctrl+S）：

              
              
// Google Apps Script - 社團問卷即時雲端同步
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var data = {};
  if (rows.length > 1) {
    for (var i = 1; i < rows.length; i++) {
      var r = rows[i];
      if (r[0]) {
        var name = String(r[0]).trim();
        data[name] = {
          name: name,
          willingness: r[1] || '',
          willingnessText: r[2] || '',
          clubs: r[3] ? String(r[3]).split(';').map(function(s){return s.trim();}) : [],
          timeSlots: r[4] ? String(r[4]).split(';').map(function(s){return s.trim();}) : [],
          otherTime: r[5] ? String(r[5]) : '',
          notes: r[6] ? String(r[6]) : '',
          updatedAt: r[7] ? String(r[7]) : ''
        };
      }
    }
  }
  return ContentService.createTextOutput(JSON.stringify({ status: 'ok', data: data }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(['姓名', '參與意願', '意向說明', '想參加社團', '出席時段', '自訂時段', '留言備註', '更新時間']);
  }
  var contents = e.postData.contents;
  var payload = JSON.parse(contents);
  var r = payload.data;
  if (!r || !r.name) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error' }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  var rows = sheet.getDataRange().getValues();
  var foundRow = -1;
  for (var i = 1; i < rows.length; i++) {
    if (String(rows[i][0]).trim() === String(r.name).trim()) {
      foundRow = i + 1;
      break;
    }
  }
  var rowData = [
    r.name,
    r.willingness || '',
    r.willingnessText || '',
    (r.clubs || []).join('; '),
    (r.timeSlots || []).join('; '),
    r.otherTime || '',
    r.notes || '',
    r.updatedAt || ''
  ];
  if (foundRow > 0) {
    sheet.getRange(foundRow, 1, 1, 8).setValues([rowData]);
  } else {
    sheet.appendRow(rowData);
  }
  return ContentService.createTextOutput(JSON.stringify({ status: 'ok' }))
    .setMimeType(ContentService.MimeType.JSON);
}
